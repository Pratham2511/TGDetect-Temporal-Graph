   330→function DashboardPage({ axisColor, gridColor, axisLineColor, chartTooltipStyle }: {
   331→  axisColor: string; gridColor: string; axisLineColor: string; chartTooltipStyle: any;
   332→}) {
   333→  const totalThreats = detectionResults.length;
   334→  const criticalCount = detectionResults.filter(d => d.severity === 'Critical').length;
   335→  const contained = detectionResults.filter(d => d.status === 'Contained').length;
   336→  const avgConfidence = Math.round(detectionResults.reduce((a, d) => a + d.confidence, 0) / detectionResults.length * 10) / 10;
   337→
   338→  return (
   339→    <div className="space-y-6">
   340→      {/* KPI Cards */}
   341→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   342→        <StatCard title="Total Detections" value={totalThreats.toString()} subtitle="Last 24 hours" icon={ShieldAlert} trend="up" trendValue="+12%" color="blue" />
   343→        <StatCard title="Critical Alerts" value={criticalCount.toString()} subtitle="Requires attention" icon={AlertTriangle} trend="up" trendValue="+3" color="red" />
   344→        <StatCard title="Threats Contained" value={contained.toString()} subtitle={`${Math.round(contained / totalThreats * 100)}% containment rate`} icon={CheckCircle2} trend="up" trendValue="+8%" color="emerald" />
   345→        <StatCard title="Avg. Confidence" value={`${avgConfidence}%`} subtitle="V16 Apex model confidence" icon={Brain} trend="stable" trendValue="0.989 F1" color="purple" />
   346→      </div>
   347→
   348→      {/* Main Grid */}
   349→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   350→        {/* Traffic Overview */}
   351→        <Card className="lg:col-span-2 bg-[var(--bg-card)] border-[var(--border-primary)]">
   352→          <CardHeader className="pb-2">
   353→            <CardTitle className="text-sm font-semibold text-[var(--text-secondary)]">Network Traffic Overview</CardTitle>
   354→            <CardDescription className="text-xs text-[var(--text-muted)]">24-hour event distribution by type (all sources fused)</CardDescription>

   356→          <CardContent>
   357→            <div className="h-72">
   358→              <ResponsiveContainer width="100%" height="100%">

   360→                  <defs>
   361→                    <linearGradient id="gradNormal" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" stopColor="#3b82f6" stopOpacity={0.3} /><stop offset="100%" stopColor="#3b82f6" stopOpacity={0} /></linearGradient>

   475→function StatCard({ title, value, subtitle, icon: Icon, trend, trendValue, color }: {
   476→  title: string; value: string; subtitle: string; icon: LucideIcon;
   477→  trend: 'up' | 'down' | 'stable'; trendValue: string; color: string;
   478→}) {
   479→  const colorMap: Record<string, { bg: string; text: string; shadow: string }> = {
   480→    blue: { bg: 'bg-blue-500/10', text: 'text-blue-400', shadow: 'shadow-blue-500/10' },
   481→    red: { bg: 'bg-red-500/10', text: 'text-red-400', shadow: 'shadow-red-500/10' },
   482→    emerald: { bg: 'bg-emerald-500/10', text: 'text-emerald-400', shadow: 'shadow-emerald-500/10' },
   483→    purple: { bg: 'bg-purple-500/10', text: 'text-purple-400', shadow: 'shadow-purple-500/10' },
   484→  };
   485→  const c = colorMap[color] || colorMap.blue;
   486→  return (
   487→    <Card className="bg-[var(--bg-card)] border-[var(--border-primary)] hover:border-[var(--border-secondary)] transition-colors">
   488→      <CardContent className="p-4">
   489→        <div className="flex items-start justify-between">
   490→          <div>

   492→            <p className="text-2xl font-bold text-[var(--text-primary)] mt-1">{value}</p>
   493→            <p className="text-[11px] text-[var(--text-muted)] mt-1">{subtitle}</p>
   494→          </div>
   495→          <div className={`w-10 h-10 rounded-lg ${c.bg} flex items-center justify-center shadow-lg ${c.shadow}`}>
   496→            <Icon className={`w-5 h-5 ${c.text}`} />
   497→          </div>
   498→        </div>
   499→        <div className="flex items-center gap-1 mt-3 text-xs">
   500→          {trend === 'up' ? <ArrowUpRight className="w-3 h-3 text-emerald-400" /> :
   501→           trend === 'down' ? <ArrowDownRight className="w-3 h-3 text-red-400" /> :
   502→           <TrendingUp className="w-3 h-3 text-[var(--text-secondary)]" />}
   503→          <span className={trend === 'up' ? 'text-emerald-400' : trend === 'down' ? 'text-red-400' : 'text-[var(--text-secondary)]'}>{trendValue}</span>
