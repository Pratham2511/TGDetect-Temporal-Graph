   130→function SourceIcon({ source }: { source: string }) {
   131→  switch (source) {
   132→    case 'DARPA TC':
   133→      return <Shield className="w-4 h-4 text-blue-400" />;
   134→    case 'UNSW-NB15':
   135→      return <Network className="w-4 h-4 text-purple-400" />;

   137→      return <Activity className="w-4 h-4 text-cyan-400" />;
   138→    default:
   139→      return <Database className="w-4 h-4 text-[var(--text-secondary)]" />;
   140→  }
   141→}
   142→
   143→function formatNumber(n: number): string {
   144→  if (n >= 1_000_000) return (n / 1_000_000).toFixed(1) + 'M';
   145→  if (n >= 1_000) return (n / 1_000).toFixed(1) + 'K';
   146→  return n.toLocaleString();
   147→}
