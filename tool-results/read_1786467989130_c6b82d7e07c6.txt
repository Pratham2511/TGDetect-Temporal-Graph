   187→function formatNumber(n: number): string {
   188→  if (n >= 1_000_000) return (n / 1_000_000).toFixed(1) + 'M';
   189→  if (n >= 1_000) return (n / 1_000).toFixed(1) + 'K';
   190→  return n.toLocaleString();
   191→}
